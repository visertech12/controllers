            const currentText = xhr.responseText;
            const lastBoundary = currentText.lastIndexOf('\n\n');
            if (lastBoundary === -1 || lastBoundary < lastLen) return;
            const raw = currentText.substring(lastLen, lastBoundary);

                                const a = document.createElement('a');

                                a.download = 'scan_results_' + new Date().getTime() + '.json';

                                a.click();
                                document.body.removeChild(a);
                                URL.revokeObjectURL(url);

                }
            });
        }; xhr.send(fd);
