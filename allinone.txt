function sys_print($msg, $type = 'success', $sleep_ms = 20) {

// Helper function to recursively clear directories
function clear_laravel_directory($dir) {
    if (!is_dir($dir)) return 0;
    
    $deleted_count = 0;
    $items = scandir($dir);
    if ($items === false) return 0;
    
    $items = array_diff($items, array('.', '..', '.gitignore'));
    
    foreach ($items as $item) {
        $path = $dir . DIRECTORY_SEPARATOR . $item;
        
        if (is_dir($path)) {
            $deleted_count += clear_laravel_directory($path);
            @rmdir($path);
        } else {
            if (@unlink($path)) {
                $deleted_count++;
            }
        }
    }
    return $deleted_count;
}

// Helper function to recursively find files
function universal_file_search($dir, $target_name, $max_depth = 15, $current_depth = 0) {
    $results = [];
    if ($current_depth > $max_depth) return $results;
    if (!@is_dir($dir)) return $results;
    
    $items = @scandir($dir);
    if ($items === false) return $results;
    
    foreach ($items as $item) {
        if ($item === '.' || $item === '..') continue;
        $path = rtrim($dir, DIRECTORY_SEPARATOR) . DIRECTORY_SEPARATOR . $item;
        
        if (is_dir($path)) {
            if ($item === 'node_modules' || $item === '.git') continue;
            $results = array_merge($results, universal_file_search($path, $target_name, $max_depth, $current_depth + 1));
        } else {
            if ($item === $target_name) {
                $results[] = $path;
            }
        }
    }
    return $results;
}
