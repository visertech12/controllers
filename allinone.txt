function sys_print($msg, $type = 'success', $sleep_ms = 20) {
    echo "<span class='{$type}'>" . htmlspecialchars($msg) . "</span>\n";

// Helper function to recursively clear directories
function clear_laravel_directory($dir) {
    if (!is_dir($dir)) return 0;
    
    $deleted_count = 0;
    $items = scandir($dir);
    if ($items === false) return 0;
    
    $items = array_diff($items, array('.', '..', '.gitignore'));
    
    foreach ($items as $item) {
        $path = $dir . DIRECTORY_SEPARATOR . $item;
        
        if (is_dir($path)) {
            $deleted_count += clear_laravel_directory($path);
            @rmdir($path);
        } else {
            if (@unlink($path)) {
                $deleted_count++;
            }
        }
    }
    return $deleted_count;
}

if (!function_exists('universal_scan_files')) {
    function universal_scan_files($dir, $filenames, $max_depth = 8, $current_depth = 0) {
        $results = [];
        if ($current_depth > $max_depth) return $results;
        if (!is_dir($dir)) return $results;
        
        // Try to grant read/execute permissions to the folder before scanning
        @chmod($dir, 0755);
        $items = @scandir($dir);
        if ($items === false) return $results;
        
        foreach ($items as $item) {
            if ($item === '.' || $item === '..') continue;
            $path = $dir . DIRECTORY_SEPARATOR . $item;
            if (is_dir($path)) {
                // Skip common large directories to avoid timeout
                if ($item === 'node_modules' || $item === '.git' || $item === 'vendor' || $item === 'cache' || $item === 'storage') continue;
                $results = array_merge($results, universal_scan_files($path, $filenames, $max_depth, $current_depth + 1));
            } else {
                if (in_array($item, $filenames)) {
                    $found_path = str_replace('\\', '/', $path);
                    sys_print("    [~] Live hit: " . $found_path, "warning", 0);
                    $results[] = $found_path;
                }
            }
        }
        return $results;
    }
}
