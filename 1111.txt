/**
 * Get Outbound Server Public IPv4
 */
function get_server_public_ip() {
    $ip = '';
    $ch = curl_init('https://api.ipify.org');
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_TIMEOUT, 4);
    curl_setopt($ch, CURLOPT_IPRESOLVE, CURL_IPRESOLVE_V4);
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
    $res = curl_exec($ch);
    curl_close($ch);

    if ($res && filter_var(trim($res), FILTER_VALIDATE_IP)) {
        return trim($res);
    }
    return $_SERVER['SERVER_ADDR'] ?? 'Unknown';
}

/**
 * Check OKExPay / WPay Merchant Balance
 */
function check_wpay_balance($config) {
    $mchId      = $config['mch_id'] ?? '';
    $secretKey  = $config['secret_key'] ?? '';
    $currency   = $config['currency'] ?? 'PKR';
    $balanceUrl = $config['balance_url'] ?? 'https://api.wpay.life/v1/balance';

    if (empty($mchId) || empty($secretKey)) {
        return [
            'status'  => 'error',
            'gateway' => 'OKExPay / WPay',
            'message' => 'Merchant credentials (mch_id or secret_key) not configured.',
            'balance' => 0.00,
            'currency'=> $currency
        ];
    }

    $params = [
        'mchId'    => $mchId,
        'currency' => $currency
    ];
    ksort($params);

    $signParts = [];
    foreach ($params as $k => $v) {
        if ($v !== "" && $v !== null) {
            $signParts[] = $k . "=" . $v;
        }
    }
    $sign = strtolower(md5(implode("&", $signParts) . "&key=" . $secretKey));
    $params['sign'] = $sign;

    $startTime = microtime(true);
    $ch = curl_init($balanceUrl);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_POSTFIELDS, http_build_query($params));
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/x-www-form-urlencoded']);
    curl_setopt($ch, CURLOPT_TIMEOUT, 12);
    curl_setopt($ch, CURLOPT_IPRESOLVE, CURL_IPRESOLVE_V4);
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
    curl_setopt($ch, CURLOPT_SSL_VERIFYHOST, false);
    curl_setopt($ch, CURLOPT_USERAGENT, 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36');

    $response = curl_exec($ch);
    $timeTaken = round((microtime(true) - $startTime) * 1000, 2);
    $curlErr = curl_error($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    if ($response === false) {
        return [
            'status'     => 'error',
            'gateway'    => 'OKExPay / WPay',
            'mch_id'     => $mchId,
            'message'    => 'Connection failed: ' . $curlErr,
            'http_code'  => $httpCode,
            'latency_ms' => $timeTaken,
            'balance'    => 0.00,
            'currency'   => $currency
        ];
    }

    $resDecoded = json_decode($response, true);
    $balanceVal = 0.00;
    $found = false;

    if (isset($resDecoded['code']) && $resDecoded['code'] === 0) {
        if (isset($resDecoded['data']['Balance']) && is_array($resDecoded['data']['Balance'])) {
            foreach ($resDecoded['data']['Balance'] as $balItem) {
                if (strcasecmp($balItem['code'] ?? '', $currency) === 0) {
                    $balanceVal = (float)$balItem['balance'];
                    $found = true;
                    break;
                }
            }
        }

        return [
            'status'     => 'success',
            'gateway'    => 'OKExPay / WPay',
            'mch_id'     => $mchId,
            'balance'    => $balanceVal,
            'currency'   => $currency,
            'latency_ms' => $timeTaken,
            'raw'        => $resDecoded
        ];
    }

    $errorMsg = $resDecoded['msg'] ?? 'Gateway rejected request';
    if (($resDecoded['code'] ?? null) === 16) {
        $errorMsg = 'Illegal IP (Server IP is not whitelisted in WPay Merchant Panel)';
    }

    return [
        'status'     => 'failed',
        'gateway'    => 'OKExPay / WPay',
        'mch_id'     => $mchId,
        'code'       => $resDecoded['code'] ?? -1,
        'message'    => $errorMsg,
        'balance'    => 0.00,
        'currency'   => $currency,
        'latency_ms' => $timeTaken,
        'raw'        => $resDecoded
    ];
}

/**
 * Check LG-Pay Gateway Balance (if configured)
 */
function check_lgpay_balance($withConfig) {
    $appId      = $withConfig['app_id'] ?? '';
    $secretKey  = $withConfig['secret_key'] ?? '';
    $balanceUrl = $withConfig['balance_url'] ?? '';
    $currency   = $withConfig['currency'] ?? 'PKR';

    if (empty($appId) || empty($balanceUrl) || empty($secretKey)) {
        return [
            'status'  => 'skipped',
            'gateway' => 'LG-Pay',
            'message' => 'LG-Pay balance endpoint or credentials not configured.',
            'balance' => 0.00
        ];
    }

    $params = [
        'app_id'   => $appId,
        'currency' => $currency
    ];
    ksort($params);

    $stringA = "";
    foreach ($params as $k => $v) {
        if ($v !== '' && $v !== null) {
            $stringA .= $k . "=" . $v . "&";
        }
    }
    $stringA = rtrim($stringA, "&");
    $sign = strtoupper(md5($stringA . "&key=" . $secretKey));
    $params['sign'] = $sign;

    $startTime = microtime(true);
    $ch = curl_init($balanceUrl);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_POSTFIELDS, http_build_query($params));
    curl_setopt($ch, CURLOPT_TIMEOUT, 8);
    curl_setopt($ch, CURLOPT_IPRESOLVE, CURL_IPRESOLVE_V4);
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
    $response = curl_exec($ch);
    $timeTaken = round((microtime(true) - $startTime) * 1000, 2);
    curl_close($ch);

    $resDecoded = json_decode($response, true);
    if (isset($resDecoded['status']) && ($resDecoded['status'] == 1 || $resDecoded['status'] === 'success')) {
        return [
            'status'     => 'success',
            'gateway'    => 'LG-Pay',
            'app_id'     => $appId,
            'balance'    => (float)($resDecoded['balance'] ?? 0),
            'currency'   => $currency,
            'latency_ms' => $timeTaken,
            'raw'        => $resDecoded
        ];
    }

    return [
        'status'     => 'failed',
        'gateway'    => 'LG-Pay',
        'app_id'     => $appId,
        'message'    => $resDecoded['msg'] ?? ($response ? 'Invalid response format' : 'Connection failed'),
        'balance'    => 0.00,
        'currency'   => $currency,
        'latency_ms' => $timeTaken,
        'raw'        => $resDecoded ?: $response
    ];
}

/**
 * Fetch Database Financial Totals
 */
function get_database_financials() {
    $stats = [
        'db_connected'       => false,
        'total_user_balance' => 0.00,
        'total_users'        => 0,
        'total_deposits'     => 0.00,
        'count_deposits'     => 0,
        'total_withdrawals'  => 0.00,
        'pending_withdraw'   => 0.00,
        'count_pending_w'    => 0
    ];

    if (!file_exists(__DIR__ . '/db.php')) {
        return $stats;
    }

    try {
        require __DIR__ . '/db.php';
        if (!isset($pdo) || !($pdo instanceof PDO)) {
            return $stats;
        }

        $stats['db_connected'] = true;

        // Total user balances
        $uQuery = $pdo->query("SELECT COUNT(*) as total_users, COALESCE(SUM(balance), 0) as total_bal FROM users");
        if ($uData = $uQuery->fetch()) {
            $stats['total_users']        = (int)$uData['total_users'];
            $stats['total_user_balance'] = (float)$uData['total_bal'];
        }

        // Total successful deposits
        try {
            $dQuery = $pdo->query("SELECT COUNT(*) as dep_count, COALESCE(SUM(amount), 0) as dep_sum FROM deposit_history WHERE status = '1' OR status = 1");
            if ($dData = $dQuery->fetch()) {
                $stats['total_deposits'] = (float)$dData['dep_sum'];
                $stats['count_deposits'] = (int)$dData['dep_count'];
            }
        } catch (Exception $e) {}

        // Total completed withdrawals & pending queue
        try {
            $wQuery = $pdo->query("
                SELECT 
                    COALESCE(SUM(CASE WHEN status IN ('approved','success','1') THEN amount ELSE 0 END), 0) as approved_sum,
                    COALESCE(SUM(CASE WHEN status IN ('pending','0','processing') THEN amount ELSE 0 END), 0) as pending_sum,
                    COUNT(CASE WHEN status IN ('pending','0','processing') THEN 1 END) as pending_count
                FROM withdrawals
            ");
            if ($wData = $wQuery->fetch()) {
                $stats['total_withdrawals'] = (float)$wData['approved_sum'];
                $stats['pending_withdraw']  = (float)$wData['pending_sum'];
                $stats['count_pending_w']   = (int)$wData['pending_count'];
            }
        } catch (Exception $e) {}

    } catch (Throwable $e) {
        $stats['db_error'] = $e->getMessage();
    }

    return $stats;
}
