<?php
// Path to your compressed .gz log file
$gzFilePath = '/home/sadevpk/logs/work.sadev.pk-ssl_log-Jul-2026.gz';

// Path where the filtered text file will be saved
$outputFilePath = __DIR__ . '/php_requests_output.txt';

// Check if the gz file exists
if (!file_exists($gzFilePath)) {
    die("Error: Log file not found at $gzFilePath");
}

// Open the gz file for reading
$gz = gzopen($gzFilePath, 'rb');
if (!$gz) {
    die("Error: Could not open the gzip file.");
}

// Open the output text file for writing
$outputFile = fopen($outputFilePath, 'wb');
if (!$outputFile) {
    gzclose($gz);
    die("Error: Could not create output text file.");
}

$matchedCount = 0;

// Read the gzipped file line by line
while (!gzeof($gz)) {
    $line = gzgets($gz);
    
    // Filter condition: Check if the line contains '.php' (case-insensitive)
    if ($line !== false && stripos($line, '.php') !== false) {
        fwrite($outputFile, $line);
        $matchedCount++;
    }
}

// Close resources
gzclose($gz);
fclose($outputFile);

echo "Success! Filtered $matchedCount PHP request lines into: <strong>$outputFilePath</strong>";
?>
