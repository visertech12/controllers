<?php
// Path to your compressed .gz log file
$gzFilePath = '/home/sadevpk/logs/work.sadev.pk-ssl_log-Jul-2026.gz';

// Check if file exists
if (!file_exists($gzFilePath)) {
    die("Error: Log file not found at $gzFilePath");
}

// Get the file name for the download prompt
$fileName = basename($gzFilePath);

// Set headers to force download without unzipping
header('Content-Description: File Transfer');
header('Content-Type: application/octet-stream');
header('Content-Disposition: attachment; filename="' . $fileName . '"');
header('Expires: 0');
header('Cache-Control: must-revalidate');
header('Pragma: public');
header('Content-Length: ' . filesize($gzFilePath));

// Clear output buffer and flush file to user
flush();
readfile($gzFilePath);
exit;
?>
